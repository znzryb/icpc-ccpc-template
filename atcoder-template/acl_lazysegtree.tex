		在 \texttt{segtree} 的幺半群 $(S,op,e)$ 上再定义作用幺半群 $(F,composition,id)$，即可支持区间修改、区间查询和树上二分，单次操作均为 $O(\log n)$。

		\textbf{五件套约束}：\texttt{op} 必须满足结合律；\texttt{e} 是其单位元；\texttt{mapping(f,x)} 表示把作用 $f$ 施加到 $x$；\texttt{composition(f,g)} 表示“先 $g$ 后 $f$”；\texttt{id} 是恒等作用。作用还必须满足对 \texttt{op} 的分配律。

		\textbf{精简写法}：\texttt{up(k)} 写成 \texttt{d[k] = mapping(lz[k], op(左, 右))}，即不变式「节点值 = 自身标记作用在两儿子之积上」，所以区间两端的下推 / 上拉不必像 ACL 那样判断边界是否对齐，无条件做即可；代价是每次上拉多一次 \texttt{mapping}，复杂度不变。

		\textbf{最大的坑}：\texttt{mapping} 拿不到区间边界。凡是更新公式需要长度，都要把 \texttt{len} 存进 $S$，让它随 \texttt{op} 一起合并。

		\textbf{接口}：除 \texttt{segtree} 的 \texttt{set/get/prod/all\_prod/max\_right/min\_left} 外，新增闭区间 \texttt{apply(l,r,f)}；单点修改写 \texttt{apply(p,p,f)}（精简版删了 ACL 的 \texttt{apply(p,f)} 重载）。因为查询前可能需要下推标记，\texttt{get/prod} 与两个二分接口不再是 \texttt{const}。

		\subsubsection{模板本体}

		\begin{minted}{cpp}
/* ACL atcoder/lazysegtree 精简移植 + 闭区间
 * lazy_segtree(n) 合法位置 0..n; 区间 [l, r] 左闭右闭
 * 单点修改写 apply(p, p, f); 空区间写 r == l - 1
 * 不变式 d[k] = mapping(lz[k], 两儿子之积), 故下推/上拉可无条件做
 */
template<class S, S (*op)(S, S), S (*e)(),
         class F, S (*mapping)(F, S),
         F (*composition)(F, F), F (*id)()>
struct lazy_segtree {
    int n, sz = 1, lg = 0;
    vector<S> d;
    vector<F> lz;
    lazy_segtree(int n = 0) : lazy_segtree(vector<S>(n + 1, e())) {}
    lazy_segtree(const vector<S> &v) : n(v.size()) {
        while (sz < n) sz *= 2, lg++;
        d.assign(2 * sz, e()), lz.assign(sz, id());
        copy(v.begin(), v.end(), d.begin() + sz);
        for (int i = sz - 1; i; i--) up(i);
    }
    void up(int k) { d[k] = mapping(lz[k], op(d[2 * k], d[2 * k + 1])); }
    void ap(int k, F f) {
        d[k] = mapping(f, d[k]);
        if (k < sz) lz[k] = composition(f, lz[k]);
    }
    void push(int k) { ap(2 * k, lz[k]), ap(2 * k + 1, lz[k]), lz[k] = id(); }
    void set(int p, S x) {
        p += sz;
        for (int i = lg; i; i--) push(p >> i);
        d[p] = x;
        for (int i = 1; i <= lg; i++) up(p >> i);
    }
    S get(int p) { return prod(p, p); }
    S all_prod() { return d[1]; }
    S prod(int l, int r) {
        S sl = e(), sr = e();
        if (l > r) return sl;
        l += sz, r += sz + 1;
        for (int i = lg; i; i--) push(l >> i), push((r - 1) >> i);
        for (; l < r; l >>= 1, r >>= 1) {
            if (l & 1) sl = op(sl, d[l++]);
            if (r & 1) sr = op(d[--r], sr);
        }
        return op(sl, sr);
    }
    void apply(int l, int r, F f) {
        if (l > r) return;
        l += sz, r += sz + 1;
        for (int i = lg; i; i--) push(l >> i), push((r - 1) >> i);
        for (int a = l, b = r; a < b; a >>= 1, b >>= 1) {
            if (a & 1) ap(a++, f);
            if (b & 1) ap(--b, f);
        }
        for (int i = 1; i <= lg; i++) up(l >> i), up((r - 1) >> i);
    }
    template<class G> int max_right(int l, G g) {
        if (l == n) return n - 1;
        l += sz;
        for (int i = lg; i; i--) push(l >> i);
        S sm = e();
        do {
            while (l % 2 == 0) l >>= 1;
            if (!g(op(sm, d[l]))) {
                while (l < sz) {
                    push(l), l *= 2;
                    if (g(op(sm, d[l]))) sm = op(sm, d[l++]);
                }
                return l - sz - 1;
            }
            sm = op(sm, d[l++]);
        } while ((l & -l) != l);
        return n - 1;
    }
    template<class G> int min_left(int r, G g) {
        if (++r == 0) return 0;
        r += sz;
        for (int i = lg; i; i--) push((r - 1) >> i);
        S sm = e();
        do {
            r--;
            while (r > 1 && r % 2) r >>= 1;
            if (!g(op(d[r], sm))) {
                while (r < sz) {
                    push(r), r = 2 * r + 1;
                    if (g(op(d[r], sm))) sm = op(d[r--], sm);
                }
                return r + 1 - sz;
            }
            sm = op(d[r], sm);
        } while ((r & -r) != r);
        return 0;
    }
};
		\end{minted}

			\subsubsection{示例：仿射修改 + 一至三次幂和}

			用 $x\leftarrow ax+b$ 统一表达乘法、加法与赋值：乘 $v$ 是 \texttt{\{v,0\}}，加 $v$ 是 \texttt{\{1,v\}}，赋值 $v$ 是 \texttt{\{0,v\}}。节点同时维护 $\sum x$、$\sum x^2$、$\sum x^3$。

			\begin{minted}{cpp}
// ACL_EXAMPLE: lazy_affine_moments
const ll MOD = 998244353;

ll mul_mod(ll a, ll b) {
    return (ll)((__int128)a * b % MOD);
}

struct S {
    ll s1, s2, s3;
    int len;
};
struct F {
    ll mul, add; // x <- mul * x + add
};

S op(S x, S y) {
    return {(x.s1 + y.s1) % MOD,
            (x.s2 + y.s2) % MOD,
            (x.s3 + y.s3) % MOD,
            x.len + y.len};
}
S e() { return {0, 0, 0, 0}; }

S mapping(F f, S x) {
    ll m = f.mul, a = f.add;
    ll m2 = mul_mod(m, m), m3 = mul_mod(m2, m);
    ll a2 = mul_mod(a, a), a3 = mul_mod(a2, a);
    ll old1 = x.s1, old2 = x.s2, old3 = x.s3;
    ll n = x.len % MOD;
    ll s1 = (mul_mod(m, old1) + mul_mod(a, n)) % MOD;
    ll s2 = (mul_mod(m2, old2)
           + mul_mod(2 * mul_mod(m, a) % MOD, old1)
           + mul_mod(a2, n)) % MOD;
    ll s3 = (mul_mod(m3, old3)
           + mul_mod(3 * mul_mod(m2, a) % MOD, old2)
           + mul_mod(3 * mul_mod(m, a2) % MOD, old1)
           + mul_mod(a3, n)) % MOD;
    return {s1, s2, s3, x.len};
}

F composition(F f, F g) {
    // 先 g 后 f
    return {mul_mod(f.mul, g.mul),
            (mul_mod(f.mul, g.add) + f.add) % MOD};
}
F id() { return {1, 0}; }

using SegMoments = lazy_segtree<S, op, e, F,
        mapping, composition, id>;

// ACL_EXAMPLE: lazy_assign_add_stats
struct S {
    ll sum, mn, mx;
    int mx_pos, left, len;
};
struct F {
    bool has_assign;
    ll assign, add;
};

S op(S a, S b) {
    if (a.len == 0) return b;
    if (b.len == 0) return a;
    int pos = a.mx >= b.mx ? a.mx_pos : b.mx_pos;
    return {a.sum + b.sum, min(a.mn, b.mn),
            max(a.mx, b.mx), pos, a.left,
            a.len + b.len};
}
S e() { return {0, 0, 0, 0, 0, 0}; }

S mapping(F f, S x) {
    if (x.len == 0) return x;
    if (f.has_assign) {
        ll v = f.assign + f.add;
        return {v * x.len, v, v,
                x.left, x.left, x.len};
    }
    x.sum += f.add * x.len;
    x.mn += f.add;
    x.mx += f.add;
    return x;
}

F composition(F f, F g) {
    if (f.has_assign) return f;
    g.add += f.add;
    return g;
}
F id() { return {false, 0, 0}; }

    ll mn, weight;
    bool empty;
};

S op(S a, S b) {
    if (a.empty) return b;
    if (b.empty) return a;

S mapping(F f, S x) {
    if (!x.empty) x.mn += f;
    return x;
}
F composition(F f, F g) { return f + g; }
