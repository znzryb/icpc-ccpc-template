		维护满足\textbf{结合律}的二元运算 \texttt{op} 与其\textbf{单位元} \texttt{e}（即幺半群 monoid）上的序列，支持单点赋值、区间求积与线段树上二分，单次操作均为 $O(\log n)$。

		\textbf{接口一览}（区间一律 $[l,r]$ 左闭右闭；\texttt{seg(n)} 的合法位置是 $0\dots n$，共 $n+1$ 个叶子）：

		\begin{itemize}
			\item \texttt{segtree<S, op, e> seg(n)}：全部初始化为 \texttt{e()}。
			\item \texttt{segtree<S, op, e> seg(v)}：由 \texttt{vector<S>} 在 $O(n)$ 内建树。
			\item \texttt{set(p,x)} / \texttt{get(p)}：单点赋值 / 单点读取。
			\item \texttt{prod(l,r)} / \texttt{all\_prod()}：闭区间 / 全区间求积；$r=l-1$ 表示空区间。
			\item \texttt{max\_right(l,f)}：返回最大的 $r$，使 $f(op(a_l,\dots,a_r))$ 为真；取不到时返回 $l-1$。
			\item \texttt{min\_left(r,f)}：返回最小的 $l$，使 $f(op(a_l,\dots,a_r))$ 为真；取不到时返回 $r+1$。
		\end{itemize}

		\textbf{注意}：\texttt{op} 与 \texttt{e} 必须是自由函数或函数指针；\texttt{e()} 必须是真单位元。\texttt{max\_right}/\texttt{min\_left} 的谓词必须满足 \texttt{f(e()) == true}，且沿扩展方向只能从真变假一次。

		\subsubsection{模板本体}

		\begin{minted}{cpp}
/* AtCoder Library v1.6 atcoder/segtree 自包含移植 + 闭区间
 * segtree(n) 合法位置 0..n; 区间 [l, r] 左闭右闭
 * 1-based 只碰 1..n; 0-based 只碰 0..n-1
 * 从 vector 建树时 v[i] 对应位置 i, 长度由 vector 决定
 * 空区间写 r == l - 1, prod 返回 e()
 */
template<class S, S (*op)(S, S), S (*e)()>
struct segtree {
public:
    segtree() : segtree(0) {}

    explicit segtree(int n)
        : segtree(vector<S>(n + 1, e())) {}

    explicit segtree(const vector<S> &v)
        : _n(int(v.size())) {
        size = (int) bit_ceil((unsigned) _n);
        log = countr_zero((unsigned) size);
        d = vector<S>(2 * size, e());
        for (int i = 0; i < _n; i++)
            d[size + i] = v[i];
        for (int i = size - 1; i >= 1; i--)
            update(i);
    }

    void set(int p, S x) {
        assert(0 <= p && p < _n);
        p += size;
        d[p] = x;
        for (int i = 1; i <= log; i++)
            update(p >> i);
    }

    S get(int p) const {
        assert(0 <= p && p < _n);
        return d[p + size];
    }

    S prod(int l, int r) const {
        r++;
        assert(0 <= l && l <= r && r <= _n);
        S sml = e(), smr = e();
        l += size, r += size;
        while (l < r) {
            if (l & 1) sml = op(sml, d[l++]);
            if (r & 1) smr = op(d[--r], smr);
            l >>= 1, r >>= 1;
        }
        return op(sml, smr);
    }

    S all_prod() const { return d[1]; }

    template<class F>
    int max_right(int l, F f) const {
        assert(0 <= l && l <= _n);
        assert(f(e()));
        if (l == _n) return _n - 1;
        l += size;
        S sm = e();
        do {
            while (l % 2 == 0) l >>= 1;
            if (!f(op(sm, d[l]))) {
                while (l < size) {
                    l = 2 * l;
                    if (f(op(sm, d[l]))) {
                        sm = op(sm, d[l]);
                        l++;
                    }
                }
                return (l - size) - 1;
            }
            sm = op(sm, d[l]);
            l++;
        } while ((l & -l) != l);
        return _n - 1;
    }

    template<class F>
    int min_left(int r, F f) const {
        r++;
        assert(0 <= r && r <= _n);
        assert(f(e()));
        if (r == 0) return 0;
        r += size;
        S sm = e();
        do {
            r--;
            while (r > 1 && (r % 2)) r >>= 1;
            if (!f(op(d[r], sm))) {
                while (r < size) {
                    r = 2 * r + 1;
                    if (f(op(d[r], sm))) {
                        sm = op(d[r], sm);
                        r--;
                    }
                }
                return r + 1 - size;
            }
            sm = op(d[r], sm);
        } while ((r & -r) != r);
        return 0;
    }

private:
    int _n, size, log;
    vector<S> d;

    static unsigned bit_ceil(unsigned n) {
        unsigned x = 1;
        while (x < n) x *= 2;
        return x;
    }

    static int countr_zero(unsigned n) {
        return __builtin_ctz(n);
    }

    void update(int k) {
        d[k] = op(d[2 * k], d[2 * k + 1]);
    }
};
		\end{minted}

// ACL_EXAMPLE: seg_max_subarray
struct S {
    ll sum, pre, suf, best;
    bool empty;
};

S op(S a, S b) {
    if (a.empty) return b;
    if (b.empty) return a;

        a.sum + b.sum,
        max(a.pre, a.sum + b.pre),
        max(b.suf, b.sum + a.suf),
        max({a.best, b.best, a.suf + b.pre}),
        false
    };
}

S make_leaf(ll x) { return {x, x, x, x, false}; }

			\subsubsection{示例：第 $k$ 个 1 与左右边界}

			当叶子只有 $0/1$ 时，前缀计数具有单调性。\texttt{max\_right} 找到“计数仍小于 $k$”的最后位置，再加一就是第 $k$ 个 1。

			\begin{minted}{cpp}
// ACL_EXAMPLE: seg_kth_boundary
using S = ll;
S op(S a, S b) { return a + b; }
S e() { return 0; }
using SegCount = segtree<S, op, e>;

int kth_one(const SegCount &seg, ll k) {
    if (k <= 0 || seg.all_prod() < k) return -1;
    return seg.max_right(1, [&](S cnt) {
        return cnt < k;
    }) + 1;
}

int farthest_right(const SegCount &seg, int l, ll x) {
    // 元素非负时，区间和 <= x 才沿右扩单调。
    return seg.max_right(l, [&](S sum) {
        return sum <= x;
    });
}

int farthest_left(const SegCount &seg, int r, ll x) {
    int l = seg.min_left(r, [&](S sum) {
        return sum <= x;
    });
    return max(1, l); // 1-based 时排除额外的 0 号单位元
}
			\end{minted}

			\subsubsection{示例：含负数时找首个前缀和 $\ge x$}

			有负数时不能拿区间和判定单调性；节点必须维护“这段能达到的最大前缀和” \texttt{pre}。若累计信息的 \texttt{pre < x}，说明还没遇到答案。

			\begin{minted}{cpp}
// ACL_EXAMPLE: seg_negative_prefix
struct S {
    ll sum, pre;
    bool empty;
};

S op(S a, S b) {
    if (a.empty) return b;
    if (b.empty) return a;
    return {a.sum + b.sum,
            max(a.pre, a.sum + b.pre), false};
}

S e() { return {0, 0, true}; }
S make_leaf(ll x) { return {x, x, false}; }
using SegPrefix = segtree<S, op, e>;

int first_prefix_at_least(
        const SegPrefix &seg, int l, int n, ll x) {
    if (seg.prod(l, n).pre < x) return -1;
    int last_bad = seg.max_right(l, [&](S cur) {
        return cur.empty || cur.pre < x;
    });
    return last_bad + 1;
}
